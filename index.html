/* Updated CSS media query for mobile hamburger menu */ @media (max-width: 768px) { .hamburger-menu { width: 100%; z-index: 999; gap: 20px; padding: 10px; } } /* Improved JavaScript function to close menu on link click */ document.querySelectorAll('.nav-link').forEach(link => { link.addEventListener('click', () => { toggleMenu(); document.querySelector('.hamburger-menu').setAttribute('aria-expanded', 'false'); }); }); function toggleMenu() { const menu = document.querySelector('.hamburger-menu'); const isExpanded = menu.getAttribute('aria-expanded') === 'true'; menu.setAttribute('aria-expanded', !isExpanded); menu.classList.toggle('active'); }
